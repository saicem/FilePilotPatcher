cmake_minimum_required(VERSION 3.22)
project(FilePilotDialogs LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8)
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde32.c vendor/minhook/src/hde/hde64.c)
add_library(FPilotDialogs SHARED adapter.cpp modern_dialog.cpp client.cpp agent_shared.cpp attachment.cpp child_delivery.cpp adapter.def)
target_link_libraries(FPilotDialogs PRIVATE minhook ole32 shell32 shlwapi uuid comdlg32 advapi32 psapi propsys)
target_compile_definitions(FPilotDialogs PRIVATE _WIN32_WINNT=0x0A00)
add_library(FPilotDialogCom SHARED com_server.cpp modern_dialog.cpp client.cpp agent_shared.cpp com_server.def)
target_link_libraries(FPilotDialogCom PRIVATE ole32 shell32 shlwapi uuid advapi32 propsys)
target_compile_definitions(FPilotDialogCom PRIVATE _WIN32_WINNT=0x0A00)
add_library(FPilotDialogWinRT SHARED winrt_server.cpp client.cpp agent_shared.cpp winrt_server.def)
target_link_libraries(FPilotDialogWinRT PRIVATE windowsapp ole32 shell32 shlwapi uuid advapi32)
target_compile_definitions(FPilotDialogWinRT PRIVATE _WIN32_WINNT=0x0A00)
set_property(TARGET FPilotDialogWinRT PROPERTY CXX_STANDARD 20)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(FPilotDialogCom PROPERTIES OUTPUT_NAME FPilotDialogCom64)
    set_target_properties(FPilotDialogWinRT PROPERTIES OUTPUT_NAME FPilotDialogWinRT64)
else()
    set_target_properties(FPilotDialogCom PROPERTIES OUTPUT_NAME FPilotDialogCom32)
    set_target_properties(FPilotDialogWinRT PROPERTIES OUTPUT_NAME FPilotDialogWinRT32)
endif()
add_executable(FPilotDialogHost WIN32 agent_host.cpp attachment.cpp agent_shared.cpp)
target_link_libraries(FPilotDialogHost PRIVATE advapi32 ole32 shell32 shlwapi uuid psapi)
target_compile_definitions(FPilotDialogHost PRIVATE _WIN32_WINNT=0x0A00)
add_library(FPilotDialogBootstrap SHARED agent_bootstrap.cpp agent_bootstrap.def)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(FPilotDialogExternal WIN32 external_agent.cpp external_dialog.cpp client.cpp agent_shared.cpp)
    target_link_libraries(FPilotDialogExternal PRIVATE advapi32 ole32 oleaut32 shell32 shlwapi uuid comctl32 uiautomationcore)
    target_compile_definitions(FPilotDialogExternal PRIVATE _WIN32_WINNT=0x0A00)
    add_executable(FPilotDialogAgent WIN32 agent.cpp agent_shared.cpp)
    target_link_libraries(FPilotDialogAgent PRIVATE advapi32 ole32 shell32 shlwapi uuid comctl32)
    target_compile_definitions(FPilotDialogAgent PRIVATE _WIN32_WINNT=0x0A00)
    set_target_properties(FPilotDialogHost PROPERTIES OUTPUT_NAME FPilotDialogHost64)
    set_target_properties(FPilotDialogBootstrap PROPERTIES OUTPUT_NAME FPilotDialogBootstrap64)
else()
    set_target_properties(FPilotDialogHost PROPERTIES OUTPUT_NAME FPilotDialogHost32)
    set_target_properties(FPilotDialogBootstrap PROPERTIES OUTPUT_NAME FPilotDialogBootstrap32)
endif()
add_executable(FPilotDialogLauncher launcher.cpp client.cpp)
target_link_libraries(FPilotDialogLauncher PRIVATE shlwapi psapi ole32 shell32 uuid)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_library(FPilotPickerRuntime SHARED picker_runtime.cpp native_ui.cpp)
    target_link_libraries(FPilotPickerRuntime PRIVATE minhook ole32 shell32 shlwapi uuid comctl32 imm32)
else()
    set_target_properties(FPilotDialogs PROPERTIES OUTPUT_NAME FPilotDialogs32)
    set_target_properties(FPilotDialogLauncher PROPERTIES OUTPUT_NAME FPilotDialogLauncher32)
endif()
